if(STRATA_HY4_CUDA)
  if(STRATA_ENABLE_HIP OR STRATA_ENABLE_SYCL)
    message(FATAL_ERROR "STRATA_HY4_CUDA is CUDA only; leave HIP and SYCL off")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    message(FATAL_ERROR "Set CMAKE_CUDA_ARCHITECTURES for the Hy4 CUDA build (e.g. 86 for A6000)")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(TARGET strata_kernels)
    set(_hy4_kernels strata_kernels)
  else()
    add_library(strata_hy4_cuda_kernels STATIC ../kernels/cuda/native_mmvq.cu ../kernels/cuda/iq_kernels.cu pdl_eager.cpp)
    target_link_libraries(strata_hy4_cuda_kernels PRIVATE CUDA::cudart)
    target_include_directories(strata_hy4_cuda_kernels PRIVATE ${PROJECT_SOURCE_DIR}/include ${PROJECT_SOURCE_DIR}/third_party/ggml)
    target_compile_definitions(strata_hy4_cuda_kernels PRIVATE STRATA_VERIFY_MAX_T=8)
    target_compile_options(strata_hy4_cuda_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math;--expt-relaxed-constexpr>)
    set_target_properties(strata_hy4_cuda_kernels PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
    set(_hy4_kernels strata_hy4_cuda_kernels)
  endif()
  target_sources(strata_hy4 PRIVATE cuda.cpp)
  target_link_libraries(strata_hy4 PUBLIC ${_hy4_kernels} CUDA::cudart)
  target_compile_definitions(strata-hy4 PRIVATE STRATA_HY4_CUDA=1)
endif()
